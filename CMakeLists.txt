# Compilación de PC: motor + simulador + pruebas.
# (El firmware del reloj está en firmware/ y se compila con ESP-IDF.)
cmake_minimum_required(VERSION 3.16)
project(mascota_pc C)
set(CMAKE_C_STANDARD 99)

add_subdirectory(core)

add_executable(mascota_sim sim/main.c)
target_link_libraries(mascota_sim PRIVATE pet_core)

# Catálogo de sonidos para el editor: ./mascota_sonidos docs
add_executable(mascota_sonidos sim/preview/sonidos.c)
target_link_libraries(mascota_sonidos PRIVATE pet_core)

# Vista previa de la interfaz (opcional): cmake -DLVGL_DIR=/ruta/a/lvgl-9.4 ...
set(LVGL_DIR "" CACHE PATH "Ruta a LVGL 9.4 para compilar la vista previa de la interfaz")
if(LVGL_DIR)
    file(GLOB_RECURSE LVGL_SRC ${LVGL_DIR}/src/*.c)
    add_library(lvgl_host STATIC ${LVGL_SRC})
    target_include_directories(lvgl_host PUBLIC ${LVGL_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/sim/preview)
    target_compile_definitions(lvgl_host PUBLIC LV_CONF_INCLUDE_SIMPLE)
    add_executable(mascota_preview sim/preview/preview.c firmware/main/ui.c firmware/main/art.c)
    target_include_directories(mascota_preview PRIVATE firmware/main)
    target_link_libraries(mascota_preview PRIVATE lvgl_host pet_core m)
    # v89: ui.c desvía dos funciones de LVGL (ver los __wrap_ en ui.c)
    target_link_options(mascota_preview PRIVATE -Wl,--wrap=lv_obj_set_local_style_prop -Wl,--wrap=lv_obj_move_to_index)
    add_executable(mascota_casa sim/preview/casa_sheet.c firmware/main/art.c)
    target_include_directories(mascota_casa PRIVATE firmware/main)
    target_link_libraries(mascota_casa PRIVATE lvgl_host pet_core m)
    add_executable(mascota_fondo sim/preview/fondo_sheet.c firmware/main/art.c)
    target_include_directories(mascota_fondo PRIVATE firmware/main)
    target_link_libraries(mascota_fondo PRIVATE lvgl_host pet_core m)
    add_executable(mascota_historia sim/preview/historia_sheet.c firmware/main/ui.c firmware/main/art.c)
    target_include_directories(mascota_historia PRIVATE firmware/main)
    target_link_libraries(mascota_historia PRIVATE lvgl_host pet_core m)
    target_link_options(mascota_historia PRIVATE -Wl,--wrap=lv_obj_set_local_style_prop -Wl,--wrap=lv_obj_move_to_index)
    add_executable(mascota_png sim/preview/mascota_png.c firmware/main/art.c)
    target_include_directories(mascota_png PRIVATE firmware/main)
    target_link_libraries(mascota_png PRIVATE lvgl_host pet_core m)
    add_executable(mascota_arte sim/preview/art_sheet.c firmware/main/art.c)
    target_include_directories(mascota_arte PRIVATE firmware/main)
    target_link_libraries(mascota_arte PRIVATE lvgl_host pet_core m)
    add_executable(mascota_editor_datos sim/preview/editor_datos.c firmware/main/art.c)
    target_include_directories(mascota_editor_datos PRIVATE firmware/main)
    target_link_libraries(mascota_editor_datos PRIVATE lvgl_host pet_core m)
    add_executable(mascota_iconos sim/preview/catalogo.c firmware/main/art.c)
    target_include_directories(mascota_iconos PRIVATE firmware/main)
    target_link_libraries(mascota_iconos PRIVATE lvgl_host pet_core m)
    # criaturas dibujadas por el reloj, para comparar con la vista previa del editor (editor/pruebas/prueba_criaturas.js)
    add_executable(mascota_criaturas_ref sim/preview/criaturas_ref.c firmware/main/art.c)
    target_include_directories(mascota_criaturas_ref PRIVATE firmware/main)
    target_link_libraries(mascota_criaturas_ref PRIVATE lvgl_host pet_core m)
endif()

enable_testing()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE pet_core)
add_test(NAME motor COMMAND test_core)
